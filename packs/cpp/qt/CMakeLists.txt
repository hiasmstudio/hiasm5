cmake_minimum_required(VERSION 3.21)
project(HiAsmQtForm LANGUAGES CXX)

file(TO_CMAKE_PATH "${HIASM_SOURCE}" HIASM_SOURCE)
if(HIASM_LIBRARY)
    add_library(program SHARED "${HIASM_SOURCE}")
    set_target_properties(program PROPERTIES PREFIX "" SUFFIX "${HIASM_SUFFIX}")
    if(HIASM_COM)
        target_link_libraries(program PRIVATE ole32 advapi32 uuid)
    endif()
elseif(HIASM_WORKER)
    add_executable(program WIN32 "${HIASM_SOURCE}")
elseif(HIASM_CONSOLE)
    add_executable(program "${HIASM_SOURCE}")
else()
    find_package(Qt6 6.9 REQUIRED COMPONENTS Widgets)
    qt_add_executable(program WIN32 "${HIASM_SOURCE}")
    if(HIASM_CORE)
        set(core_dir "${CMAKE_CURRENT_SOURCE_DIR}/runtime")
        if(NOT EXISTS "${core_dir}/core_widgets.cpp")
            set(core_dir "${CMAKE_CURRENT_SOURCE_DIR}/../../../src/qt")
        endif()
        target_sources(program PRIVATE "${core_dir}/core_widgets.cpp" "${core_dir}/core_application.cpp")
    endif()
    set_target_properties(program PROPERTIES MACOSX_BUNDLE FALSE)
    target_link_libraries(program PRIVATE Qt6::Widgets)
    if(HIASM_WEBENGINE)
        target_compile_definitions(program PRIVATE HIASM_QT_WEBENGINE)
        target_include_directories(program PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
        if(WIN32)
            include(WebView2.cmake)
            hiasm_webview2(program)
            target_sources(program PRIVATE web_widgets_win.cpp)
        else()
            find_package(Qt6 6.9 REQUIRED COMPONENTS WebEngineWidgets PrintSupport)
            target_sources(program PRIVATE web_widgets.cpp)
            target_link_libraries(program PRIVATE Qt6::WebEngineWidgets Qt6::PrintSupport)
        endif()
        if(HIASM_RUFFLE_DIR)
            target_compile_definitions(program PRIVATE HIASM_QT_RUFFLE)
            target_sources(program PRIVATE web_flash.cpp)
            file(GLOB ruffle_assets "${HIASM_RUFFLE_DIR}/*.js" "${HIASM_RUFFLE_DIR}/*.wasm"
                "${HIASM_RUFFLE_DIR}/LICENSE*")
            list(APPEND ruffle_assets "${CMAKE_CURRENT_SOURCE_DIR}/ruffle_player.html")
            set(ruffle_qrc "${CMAKE_CURRENT_BINARY_DIR}/hiasm_ruffle.qrc")
            file(WRITE "${ruffle_qrc}" "<RCC><qresource prefix=\"/hiasm/ruffle\">\n")
            foreach(asset IN LISTS ruffle_assets)
                get_filename_component(name "${asset}" NAME)
                string(REPLACE "&" "&amp;" asset "${asset}")
                string(REPLACE "<" "&lt;" asset "${asset}")
                file(APPEND "${ruffle_qrc}" "  <file alias=\"${name}\">${asset}</file>\n")
            endforeach()
            file(APPEND "${ruffle_qrc}" "</qresource></RCC>\n")
            qt_add_big_resources(ruffle_object "${ruffle_qrc}" OPTIONS --no-compress)
            target_sources(program PRIVATE ${ruffle_object})
        endif()
    endif()
endif()
target_compile_features(program PRIVATE cxx_std_17)
if(HIASM_TOOLCHAIN_PROBE)
    target_compile_definitions(program PRIVATE HIASM_TOOLCHAIN_PROBE)
endif()
if(MSVC)
    target_compile_options(program PRIVATE /utf-8)
endif()
